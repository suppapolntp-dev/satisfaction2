"use client";

import { useEffect } from "react";

const noop = () => {};

export default function TouchEnabler() {
  useEffect(() => {
    // à¹€à¸­à¸² pointerdown à¸­à¸­à¸ â€” à¸‚à¸±à¸” React 19 synthetic events
    document.addEventListener("touchstart", noop, { passive: true });
    document.body.classList.add("touch-manipulation");

    return () => {
      document.removeEventListener("touchstart", noop);
      document.body.classList.remove("touch-manipulation");
    };
  }, []);

  return null;
}

export default function DataTable<T extends Record<string, any>>({
  columns,
  data,
  actions,
  emptyMessage = "à¹„à¸¡à¹ˆà¸žà¸šà¸‚à¹‰à¸­à¸¡à¸¹à¸¥à¸—à¸µà¹ˆà¸•à¸£à¸‡à¸à¸±à¸šà¹€à¸‡à¸·à¹ˆà¸­à¸™à¹„à¸‚à¸à¸²à¸£à¸„à¹‰à¸™à¸«à¸²",

  const [sortConfig, setSortConfig] = useState<{
    key: keyof T | null;
    direction: "asc" | "desc" | null;
  }>({ key: null, direction: null });

  // à¸Ÿà¸±à¸‡à¸à¹Œà¸Šà¸±à¸™à¸ªà¸¥à¸±à¸šà¸ªà¸–à¸²à¸™à¸°à¸‚à¸­à¸‡ à¸›à¸¸à¹ˆà¸¡ Sort (à¸ªà¸¥à¸±à¸š 3 à¸ˆà¸±à¸‡à¸«à¸§à¸°: à¸™à¹‰à¸­à¸¢à¹„à¸›à¸¡à¸²à¸ -> à¸¡à¸²à¸à¹„à¸›à¸™à¹‰à¸­à¸¢ -> à¸›à¸´à¸”à¸£à¸°à¸šà¸š)
  const handleSort = (key: keyof T) => {
    let direction: "asc" | "desc" | null = "asc";
    if (sortConfig.key === key && sortConfig.direction === "asc") {
      direction = "desc";
    } else if (sortConfig.key === key && sortConfig.direction === "desc") {
      direction = null;
    }
    setSortConfig({ key: direction ? key : null, direction });
  };

  // à¸•à¸£à¸£à¸à¸°à¸­à¸±à¸ˆà¸‰à¸£à¸´à¸¢à¸°à¸›à¸£à¸°à¸¡à¸§à¸¥à¸œà¸¥à¸„à¸±à¸”à¹à¸¢à¸à¸ªà¸¥à¸±à¸šà¸¥à¸³à¸”à¸±à¸šà¸•à¸±à¸§à¸­à¸±à¸à¸©à¸£à¹à¸¥à¸°à¸•à¸±à¸§à¹€à¸¥à¸‚ (Smart Alphanumeric Sorting Engine)
  const sortedData = useMemo(() => {
    if (!sortConfig.key || !sortConfig.direction) return data;

    return [...data].sort((a, b) => {
      const aVal = a[sortConfig.key!];
      const bVal = b[sortConfig.key!];

      if (aVal === bVal) return 0;
      
      const modifier = sortConfig.direction === "asc" ? 1 : -1;
      
      // à¸à¸£à¸“à¸µà¸‚à¹‰à¸­à¸¡à¸¹à¸¥à¹€à¸›à¹‡à¸™à¸•à¸±à¸§à¹€à¸¥à¸‚à¸„à¸§à¸²à¸¡à¸ªà¸¹à¸‡à¸•à¹ˆà¸³ (Numbers)
      if (typeof aVal === "number" && typeof bVal === "number") {
        return (aVal - bVal) * modifier;
      }
      
      // à¸à¸£à¸“à¸µà¸‚à¹‰à¸­à¸¡à¸¹à¸¥à¹€à¸›à¹‡à¸™à¸•à¸±à¸§à¸­à¸±à¸à¸©à¸£à¸‚à¹‰à¸­à¸„à¸§à¸²à¸¡à¸—à¸±à¹ˆà¸§à¹„à¸› (Strings)
      return String(aVal).localeCompare(String(bVal)) * modifier;
    });
  }, [data, sortConfig]);

  return (
    <div className="overflow-x-auto w-full">
      <table className="w-full text-left border-collapse">
        <thead>

                key={idx}
                className={`py-3 px-4 ${col.key === "actions" ? "text-right w-16" : ""}`}

                {/* à¸Šà¸´à¹‰à¸™à¸—à¸µà¹ˆ 1: à¸•à¸£à¸§à¸ˆà¹€à¸Šà¹‡à¸à¹€à¸‡à¸·à¹ˆà¸­à¸™à¹„à¸‚à¸«à¸²à¸à¹€à¸¥à¸™à¸™à¸µà¹‰à¹„à¸”à¹‰à¸£à¸±à¸šà¸ªà¸´à¸—à¸˜à¸´à¹Œ sortable à¹ƒà¸«à¹‰à¸‚à¸¶à¹‰à¸™à¸£à¸¹à¸›à¸›à¸¸à¹ˆà¸¡à¸ˆà¸±à¸”à¹€à¸£à¸µà¸¢à¸‡à¸¥à¸³à¸”à¸±à¸š */}
                {col.sortable && col.key !== "actions" ? (
                  <button
                    type="button"
                    onClick={() => handleSort(col.key as keyof T)}
                    className="inline-flex items-center gap-1 hover:text-blue-900 transition-colors outline-none cursor-pointer touch-manipulation font-bold"
                  >
                    {col.label}
                    <span className="material-symbols-outlined text-[14px] font-bold">
                      {sortConfig.key === col.key
                        ? sortConfig.direction === "asc"
                          ? "arrow_upward"
                          : "arrow_downward"
                        : "unfold_more"}
                    </span>
                  </button>
                ) : (
                  col.label
                )}
              </th>
            ))}
          </tr>
        </thead>
        
        <tbody className="text-xs text-neutral-700 divide-y divide-slate-100">
          {sortedData.length > 0 ? (
            sortedData.map((row, rowIdx) => (
              <tr
                key={row.id || rowIdx}
                className={`hover:bg-slate-50/80 transition-colors h-14 ${
                  rowIdx % 2 === 1 ? "bg-slate-50/30" : "bg-white"
                }`}
              >
                {columns.map((col, colIdx) => (
                  <td

                  >
                    {/* à¸à¸£à¸“à¸µà¸£à¸°à¸šà¸¸à¸„à¸µà¸¢à¹Œà¸Šà¹ˆà¸­à¸‡à¹€à¸›à¹‡à¸™ Actions à¸žà¸´à¹€à¸¨à¸© à¹ƒà¸«à¹‰à¸”à¸¶à¸‡ TableDropdown à¸ªà¸§à¸¡à¸—à¸³à¸‡à¸²à¸™à¸›à¸£à¸°à¸à¸šà¸—à¹‰à¸²à¸¢à¹à¸–à¸§ */}

                </button>

            <div className="p-3 bg-slate-50 rounded-xl border border-slate-200 flex items-start gap-2.5">
